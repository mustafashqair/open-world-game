  }

  async function fxLoad() {
    const B = 'three/addons/';
    const [pass, gt, bl, sm, sp, fx] = await Promise.all([import(B + 'postprocessing/Pass.js'), import(B + 'postprocessing/GTAOPass.js'), import(B + 'postprocessing/UnrealBloomPass.js'), import(B + 'postprocessing/SMAAPass.js'), import(B + 'postprocessing/ShaderPass.js'), import(B + 'shaders/FXAAShader.js')]);
    X.mods = { FullScreenQuad: pass.FullScreenQuad, GTAOPass: gt.GTAOPass, UnrealBloomPass: bl.UnrealBloomPass, SMAAPass: sm.SMAAPass, ShaderPass: sp.ShaderPass, FXAAShader: fx.FXAAShader };
  }
  const COMP_FS = `
    uniform sampler2D tScene, tAO; uniform float uAO, uGrade, uVig; varying vec2 vUv;
    vec3 l2s(vec3 c){ return mix(c * 12.92, 1.055 * pow(c, vec3(1.0 / 2.4)) - 0.055, step(vec3(0.0031308), c)); }
    void main(){ vec3 c = texture2D(tScene, vUv).rgb;
      #ifdef USE_AO
        c *= mix(1.0, texture2D(tAO, vUv).r, uAO);
      #endif
      vec3 s = l2s(clamp(c, 0.0, 1.0));
      vec3 sc = s * s * (3.0 - 2.0 * s); s = mix(s, sc, 0.10 * uGrade);                       // منحنى تباين خفيف
      float l = dot(s, vec3(0.2126, 0.7152, 0.0722)); s = mix(vec3(l), s, 1.0 + 0.06 * uGrade);  // تشبّع +٦٪
      s *= mix(vec3(1.0), vec3(1.025, 1.0, 0.965), uGrade);                                        // دفء خفيف
      s += vec3(0.012, 0.006, 0.0) * (1.0 - l) * uGrade;                                           // ظلال دافئة قليلاً
      vec2 q = vUv - 0.5; s *= 1.0 - uVig * dot(q, q) * 1.6;                                       // تظليل أطراف خفيف
      gl_FragColor = vec4(clamp(s, 0.0, 1.0), 1.0); }`;
  function fxBuild(tier) {
    fxFree(); const M = X.mods; if (!M || tier === 'none') { X.tier = 'none'; return; }
    renderer.getDrawingBufferSize(_v2); const w = Math.max(2, _v2.x | 0), h = Math.max(2, _v2.y | 0); X.w = w; X.h = h;

    X.lRT = new T.WebGLRenderTarget(w, h, { type: T.UnsignedByteType, depthBuffer: false });
    X.comp = new T.ShaderMaterial({ defines: tier === 'high' ? { USE_AO: 1 } : {}, uniforms: { tScene: { value: X.sRT.texture }, tAO: { value: null }, uAO: { value: 0.85 }, uGrade: { value: 1 }, uVig: { value: tier === 'mobile' ? 0.0 : 0.12 } },
      vertexShader: 'varying vec2 vUv; void main(){ vUv = uv; gl_Position = vec4(position.xy, 0.0, 1.0); }', fragmentShader: COMP_FS, depthTest: false, depthWrite: false });
    X.quad = new M.FullScreenQuad(X.comp);
    if (tier === 'high') {
      const g = X.gtao = new M.GTAOPass(scene, camera, w >> 1, h >> 1);
      g.normalRenderTarget.setSize(1, 1); g.setGBuffer(dt);           // عمق المشهد نفسه (بدون رسم المشهد مرة ثانية)، والعمود العمودي يُستنتج من العمق
      g.updateGtaoMaterial({ radius: 1.6, distanceExponent: 1.4, thickness: 1.2, scale: 1.0, samples: 12, distanceFallOff: 1.0 });
      g.updatePdMaterial({ lumaPhi: 10, depthPhi: 2, normalPhi: 3, radius: 6, rings: 2, samples: 12 });
      X.comp.uniforms.tAO.value = g.pdRenderTarget.texture;
    }
    if (tier === 'high' || tier === 'medium') { X.bloom = new M.UnrealBloomPass(new T.Vector2(w >> 1, h >> 1), tier === 'high' ? 0.32 : 0.26, 0.45, 0.9); }
    if (tier === 'high') { X.smaa = new M.SMAAPass(w, h); X.smaa.renderToScreen = true; }
    else { X.fxaa = new M.ShaderPass(M.FXAAShader); X.fxaa.material.uniforms.resolution.value.set(1 / w, 1 / h); X.fxaa.renderToScreen = true; }
    X.tier = tier;
  }
  function fxFree() {
    for (const k of ['sRT', 'lRT']) if (X[k]) { if (X[k].depthTexture) X[k].depthTexture.dispose(); X[k].dispose(); X[k] = null; }
    for (const k of ['gtao', 'bloom', 'smaa', 'fxaa']) if (X[k]) { try { X[k].dispose(); } catch (e) {} X[k] = null; }
    if (X.comp) { X.comp.dispose(); X.comp = null; } if (X.quad) { X.quad.dispose(); X.quad = null; } X.tier = '';
  }
  function fxRender() {

      g.renderPass(renderer, g.gtaoMaterial, g.gtaoRenderTarget, 0xffffff, 1.0);
      g.pdMaterial.uniforms.cameraProjectionMatrixInverse.value.copy(camera.projectionMatrixInverse);
      g.renderPass(renderer, g.pdMaterial, g.pdRenderTarget, 0xffffff, 1.0);
    }
    let src = X.sRT; try { if (window.V41) src = V41.post(X) || X.sRT; } catch (e) { warn('v41 post', e); }
    X.comp.uniforms.tScene.value = src.texture;
    if (X.bloom) X.bloom.render(renderer, null, src, 0, false);
    renderer.setRenderTarget(X.lRT); X.quad.render(renderer);
    if (X.smaa) X.smaa.render(renderer, null, X.lRT, 0, false); else X.fxaa.render(renderer, null, X.lRT, 0, false);
    renderer.setRenderTarget(null); renderer.autoClear = oldAC;
  }
  function fxTick() {

  }

  const A = { on: true, howl: null, loading: false, unlocked: false, feet: null, last: [0, 0], prevY: [0, 0], prevV: [0, 0], minY: [1e9, 1e9], t: 0, phase: -1, n: 0, surf: '' };

  function surfAt() {

  }
  function findFeet() {
    const M = MODELS.plSw; if (!M || !M.wrap) return null; const bones = [];
    M.wrap.traverse(o => { if (o.isBone && /foot|ankle/i.test(o.name) && !/toe|end|ball|tip/i.test(o.name)) bones.push(o); });

  }
